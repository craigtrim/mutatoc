# Everything builds and tests natively; no interpreter or model at any stage (craigtrim/mutatoc#1).
cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(mutatoc VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(MUTATOC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(mutatoc src/util.c src/uri.c src/literal.c src/rdf.c src/ontology.c src/data_query.c src/views.c src/match.c src/tokenize.c src/api.c vendor/cjson/cJSON.c)
target_include_directories(mutatoc PUBLIC include PRIVATE src vendor/cjson)
target_compile_definitions(mutatoc PUBLIC CJSON_HIDE_SYMBOLS)
set_target_properties(mutatoc PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(mutatoc PRIVATE /W4 /utf-8)
  target_compile_definitions(mutatoc PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(mutatoc PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(mutatoc PRIVATE m)
  if(MUTATOC_SANITIZE)
    # Undefined behavior aborts the test instead of printing and passing.
    target_compile_options(mutatoc PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(mutatoc PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(mutatoc_cli src/cli.c)
set_target_properties(mutatoc_cli PROPERTIES OUTPUT_NAME mutatoc ARCHIVE_OUTPUT_NAME mutatoc_cli_exports)
target_link_libraries(mutatoc_cli PRIVATE mutatoc)
target_include_directories(mutatoc_cli PRIVATE src vendor/cjson)
if(WIN32)
  target_link_libraries(mutatoc_cli PRIVATE shell32)
endif()
option(MUTATOC_BUILD_EXAMPLES "Build the public C API example" ON)
if(MUTATOC_BUILD_EXAMPLES)
  add_executable(mutatoc_example examples/embed.c)
  target_link_libraries(mutatoc_example PRIVATE mutatoc)
endif()

# Test drivers and the benchmark read fixtures from the source tree.
function(mutatoc_program name source)
  add_executable(${name} ${source})
  target_include_directories(${name} PRIVATE src vendor/cjson tests)
  target_link_libraries(${name} PRIVATE mutatoc)
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
endfunction()

mutatoc_program(mutatoc_bench tests/bench.c)
if(WIN32)
  target_link_libraries(mutatoc_bench PRIVATE psapi)
endif()

include(CTest)
if(BUILD_TESTING)
  if(MUTATOC_BUILD_EXAMPLES)
    add_test(NAME embedding COMMAND mutatoc_example)
  endif()
  # One-shot CLI output: --json stays compact, --jsonf indents two spaces.
  set(cli_parse --ontology ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/ontologies/animals-test.owl --input-text Dog)
  add_test(NAME cli_json COMMAND mutatoc_cli ${cli_parse} --json)
  set_tests_properties(cli_json PROPERTIES PASS_REGULAR_EXPRESSION "^\\{\"text\":\"dog\",\"tokens\":\\[")
  add_test(NAME cli_jsonf COMMAND mutatoc_cli ${cli_parse} --jsonf)
  set_tests_properties(cli_jsonf PROPERTIES PASS_REGULAR_EXPRESSION "^\\{\n  \"text\": \"dog\",\n  \"tokens\": \\[")
  add_test(NAME cli_stopwatch COMMAND mutatoc_cli ${cli_parse} --json --stopwatch)
  set_tests_properties(cli_stopwatch PROPERTIES PASS_REGULAR_EXPRESSION "\"text\":\"dog\".*Elapsed: [0-9]+\\.[0-9]+ (ms|s)")
  mutatoc_program(mutatoc_text_tests tests/test_text.c)
  add_test(NAME text COMMAND mutatoc_text_tests)
  mutatoc_program(mutatoc_tokenize tests/test_tokenize.c)
  add_test(NAME tokenize COMMAND mutatoc_tokenize)
  mutatoc_program(mutatoc_tests tests/test_native.c)
  add_test(NAME native COMMAND mutatoc_tests ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_parity tests/test_parity.c)
  add_test(NAME parity COMMAND mutatoc_parity ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_api_tests tests/test_api.c)
  add_test(NAME public_api COMMAND mutatoc_api_tests ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_punctuation tests/test_punctuation.c)
  add_test(NAME punctuation COMMAND mutatoc_punctuation ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_rdf tests/test_rdf.c)
  add_test(NAME rdf COMMAND mutatoc_rdf ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_extensions tests/test_extensions.c)
  add_test(NAME extensions COMMAND mutatoc_extensions ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_upstream tests/test_upstream.c)
  add_test(NAME upstream COMMAND mutatoc_upstream ${CMAKE_CURRENT_SOURCE_DIR})
  mutatoc_program(mutatoc_concurrency tests/test_concurrency.c)
  add_test(NAME concurrency COMMAND mutatoc_concurrency ${CMAKE_CURRENT_SOURCE_DIR})
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(mutatoc_upstream PRIVATE Threads::Threads)
    target_link_libraries(mutatoc_concurrency PRIVATE Threads::Threads)
  endif()
endif()
install(TARGETS mutatoc mutatoc_cli)
install(FILES include/mutatoc.h DESTINATION include)
