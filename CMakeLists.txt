cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(mutatoc VERSION 0.2.2 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(MUTATOC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(mutatoc src/util.c src/uri.c src/literal.c src/rdf.c src/ontology.c src/data_query.c src/views.c src/match.c src/spacy.c src/sparql.c src/api.c src/lingpatlab_text.c src/lingpatlab_parse.c src/lingpatlab.c src/lingpatlab_analyze.c src/lingpatlab_segment.c src/lingpatlab_validate.c vendor/cjson/cJSON.c)
target_include_directories(mutatoc PUBLIC include PRIVATE src vendor/cjson)
target_compile_definitions(mutatoc PUBLIC CJSON_HIDE_SYMBOLS)
set_target_properties(mutatoc PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(mutatoc PRIVATE /W4 /utf-8)
  target_compile_definitions(mutatoc PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(mutatoc PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(mutatoc PRIVATE m)
  if(MUTATOC_SANITIZE)
    target_compile_options(mutatoc PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mutatoc PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(mutatoc_cli src/cli.c)
set_target_properties(mutatoc_cli PROPERTIES OUTPUT_NAME mutatoc ARCHIVE_OUTPUT_NAME mutatoc_cli_exports)
target_link_libraries(mutatoc_cli PRIVATE mutatoc)
target_include_directories(mutatoc_cli PRIVATE src vendor/cjson)
if(WIN32)
  target_link_libraries(mutatoc_cli PRIVATE shell32)
endif()
add_custom_command(TARGET mutatoc_cli POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:mutatoc_cli>/runtime"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/runtime/spacy_worker.py"
    "$<TARGET_FILE_DIR:mutatoc_cli>/runtime/spacy_worker.py"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/runtime/sparql_worker.py"
    "$<TARGET_FILE_DIR:mutatoc_cli>/runtime/sparql_worker.py")
option(MUTATOC_BUILD_EXAMPLES "Build the public C API example" ON)
if(MUTATOC_BUILD_EXAMPLES)
  add_executable(mutatoc_example examples/embed.c)
  target_link_libraries(mutatoc_example PRIVATE mutatoc)
endif()
include(CTest)
if(BUILD_TESTING)
  if(MUTATOC_BUILD_EXAMPLES)
    add_test(NAME embedding COMMAND mutatoc_example)
  endif()
  add_executable(mutatoc_text_tests tests/test_text.c)
  target_include_directories(mutatoc_text_tests PRIVATE src vendor/cjson)
  target_link_libraries(mutatoc_text_tests PRIVATE mutatoc)
  if(MSVC)
    target_compile_options(mutatoc_text_tests PRIVATE /utf-8)
  endif()
  add_test(NAME text COMMAND mutatoc_text_tests)
  add_executable(mutatoc_tests tests/test_native.c)
  target_include_directories(mutatoc_tests PRIVATE src vendor/cjson)
  target_link_libraries(mutatoc_tests PRIVATE mutatoc)
  add_test(NAME native COMMAND mutatoc_tests ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(mutatoc_parity tests/test_parity.c)
  target_include_directories(mutatoc_parity PRIVATE src vendor/cjson)
  target_link_libraries(mutatoc_parity PRIVATE mutatoc)
  add_test(NAME parity COMMAND mutatoc_parity ${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(mutatoc_api_tests tests/test_api.c)
  target_include_directories(mutatoc_api_tests PRIVATE src vendor/cjson)
  target_link_libraries(mutatoc_api_tests PRIVATE mutatoc)
  add_test(NAME public_api COMMAND mutatoc_api_tests ${CMAKE_CURRENT_SOURCE_DIR})
  set(MUTATOC_TEST_PYTHON "" CACHE FILEPATH "Python with pinned spaCy and model, without LingPatLab for integration tests")
  if(MUTATOC_TEST_PYTHON)
    add_test(NAME punctuation COMMAND "${MUTATOC_TEST_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_punctuation.py" --exe $<TARGET_FILE:mutatoc_cli>
      --report "${CMAKE_CURRENT_BINARY_DIR}/punctuation-validation.json")
    set_tests_properties(punctuation PROPERTIES TIMEOUT 600 LABELS "integration;regression")
    add_test(NAME spacy_bridge COMMAND "${MUTATOC_TEST_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_spacy_bridge.py" --exe $<TARGET_FILE:mutatoc_cli>)
    set_tests_properties(spacy_bridge PROPERTIES TIMEOUT 600 LABELS "integration")
    add_test(NAME extensions COMMAND "${MUTATOC_TEST_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_extensions.py" --exe $<TARGET_FILE:mutatoc_cli>)
    set_tests_properties(extensions PROPERTIES TIMEOUT 120)
    add_test(NAME sparql COMMAND "${MUTATOC_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sparql.py" --exe $<TARGET_FILE:mutatoc_cli>)
    set_tests_properties(sparql PROPERTIES TIMEOUT 180)
    add_test(NAME lingpatlab COMMAND "${MUTATOC_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lingpatlab.py" --require-clean-runtime --exe $<TARGET_FILE:mutatoc_cli>)
    set_tests_properties(lingpatlab PROPERTIES TIMEOUT 180)
    add_test(NAME turtle COMMAND "${MUTATOC_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate_turtle.py" --exe $<TARGET_FILE:mutatoc_cli>)
  endif()
endif()
install(TARGETS mutatoc mutatoc_cli)
install(FILES include/mutatoc.h DESTINATION include)

install(FILES runtime/spacy_worker.py runtime/sparql_worker.py DESTINATION bin/runtime)
